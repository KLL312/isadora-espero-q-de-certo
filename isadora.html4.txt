<script>
// Criar estrelas
function criarEstrela() {
    const estrela = document.createElement("div");
    estrela.className = "estrela";
    estrela.innerHTML = "✨";
    estrela.style.left = Math.random() * 100 + "vw";
    estrela.style.top = Math.random() * 100 + "vh";
    document.body.appendChild(estrela);
}
for (let i = 0; i < 80; i++) criarEstrela();

// Criar rosas caindo
function criarRosa() {
    const rosa = document.createElement("div");
    rosa.className = "rosa";
    rosa.innerHTML = "🌹";
    rosa.style.left = Math.random() * 100 + "vw";
    rosa.style.animationDuration = (4 + Math.random() * 4) + "s";
    document.body.appendChild(rosa);
    setTimeout(() => rosa.remove(), 8000);
}
setInterval(criarRosa, 400);

// Criar corações subindo
function criarCoracao() {
    const c = document.createElement("div");
    c.className = "coracaoSubindo";
    c.innerHTML = "💖";
    c.style.left = Math.random() * 100 + "vw";
    c.style.animationDuration = (4 + Math.random() * 4) + "s";
    document.body.appendChild(c);
    setTimeout(() => c.remove(), 8000);
}
setInterval(criarCoracao, 800);
</script>
